InputManager.cs

This script handles retrieving the player's input while playing the game. The script works by storing a singleton input manager in the world. When the input manager is created, it creates a new game object of Controls. Controls is an auto-generated C# script representing the C# equivalent of the controls input map I designed in my group's game. When the input map is updated, the controls script updates with it, and I may reference the content inside my input manager.

Every MonoBehaviour game object may reference the input manager singleton to retrieve player input. Ideally, this singleton will exist in every scene so that any script may get player input. Each method in this manager represents a different type of player input. For example, GetPlayerMovement represents the player's WASD/arrow key inputs. A MonoBehaviour script may call this method to get the keys they pressed. The best part of this system is that it is entirely platform independent; the input manager handles inputs for console and PC. So this is a one-time solution for gathering player input

While making this manager, I ran into a design question: do I send out inputs by having MonoBehaviour scripts call methods, or do I broadcast events for MonoBehaviour scripts to subscribe to? I would opt for the latter if I were making the game alone. However, I had to consider my peers' event-driven code proficiency. Given how easy it is to make a null reference with event-driven code, I opted for the former solution. This keeps the input developer-friendly at the cost of scalability. This shouldn't be too big of an issue since this is only for player input. 


using UnityEngine;

public class InputManager : MonoBehaviour
{
    //Singleton
    public static InputManager Instance;

    //Player controls
    Controls controls;

    private void Awake()
    {

        //Singleton Pattern
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }
        else
        {
            Debug.LogWarning("Warning: Multiple input managers found in scene!");
            Destroy(gameObject);
        }

        //Craete player controls
        controls = new Controls();
        //Set controls to default
        EnableDefaultControls();
    }

    private void OnDestroy()
    {
        controls.Player.Disable();
        controls.UI.Disable();
    }


    //Control Schemes
    private void EnableDefaultControls()
    {
        SetControlSchemeToPlayer();
    }

    public void SetControlSchemeToUI()
    {
        controls.Player.Disable();
        controls.UI.Enable();
    }

    public void SetControlSchemeToPlayer()
    {
        controls.Player.Enable();
        controls.UI.Disable();
    }


    //Getting Player Input
    public Vector2 GetPlayerMovement()
    {
        //Debug.Log($"{controls.Player.Move.ReadValue()}");
        return controls.Player.Move.ReadValue();
    }

    public Vector2 GetPlayerLook()
    {
        //Debug.Log($"{controls.Player.Look.ReadValue()}");
        return controls.Player.Look.ReadValue();
    }

    public bool PlayerJump()
    {
        //Debug.Log($"{controls.Player.Jump.ReadValue()}");
        return controls.Player.Jump.ReadValue() == 1f ? true : false;
    }

    public bool PlayerAbility1()
    {
        return controls.Player.Ability1.ReadValue() == 1f? true : false;
    }

    public bool PlayerAbility2()
    {
        return controls.Player.Ability2.ReadValue() == 1f ? true : false;
    }

    public bool PlayerAbility3()
    {
        return controls.Player.Ability3.ReadValue() == 1f ? true : false;
    }

    public bool PlayerPause()
    {
        return controls.Player.Pause.ReadValue() == 1f? true : false;
    }

    public float GetPrimaryFireInput()
    {
        return controls.Player.PrimaryFire.ReadValue();
    }

    public float getSecondaryFireInput()
    {
        return controls.Player.SecondaryFire.ReadValue();
    }


    //Getting UI Input
    public Vector2 GetUINavigation()
    {
        return controls.UI.Navigation.ReadValue();
    }

    public bool UISelect()
    {
        return controls.UI.Select.ReadValue() == 1f? true : false;
    }

    public bool UIBack()
    {
        return controls.UI.Back.ReadValue() == 1f ? true : false;
    }
}
