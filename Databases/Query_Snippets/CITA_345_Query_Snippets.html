Library Management System Database



CREATE TABLE Author (
	AuthorID INT PRIMARY KEY, Name VARCHAR(255),
	Nationality VARCHAR(100),
	BirthYear INT
);

CREATE TABLE Book (
	BookID INT PRIMARY KEY, Title VARCHAR(255),
	AuthorID INT,
	Genre VARCHAR(100),
	PublishedYear INT,
	ISBN VARCHAR(50),
	FOREIGN KEY (AuthorID) REFERENCES Author (AuthorID)
);

CREATE TABLE BookCategory (
	BookID INT,
	CategoryID INT,
	PRIMARY KEY (BookID, CategoryID),
	FOREIGN KEY (BookID) REFERENCES Book (BookID),
	FOREIGN KEY (CategoryID) REFERENCES Category(CategoryID)
);

CREATE TABLE Borrowing (
	BorrowID INT PRIMARY KEY,
	BookID INT,
	MemberID INT,
	BorrowDate DATE,
	ReturnDate DATE,
	FOREIGN KEY (BookID) REFERENCES Book (BookID),
	FOREIGN KEY (MemberID) REFERENCES Member (MemberID)
);

CREATE TABLE Category (
	CategoryID INT PRIMARY KEY,
	CategoryName VARCHAR(100)
);

CREATE TABLE Member (
	MemberID INT PRIMARY KEY,
	FirstName VARCHAR(255),
	LastName VARCHAR(255),
	Email VARCHAR(255),
	PhoneNumber VARCHAR(50),
	MembershipDate DATE
);

CREATE TABLE Reservation (
	ReservationID INT PRIMARY KEY,
	BookID INT,
	MemberID INT,
	ReservationDate DATE,
	FOREIGN KEY (BookID) REFERENCES Book (BookID),
	FOREIGN KEY (MemberID) REFERENCES Member (MemberID)
);

CREATE TABLE Review (
	ReviewID INT PRIMARY KEY,
	BookID INT,
	MemberID INT,
	Rating INT,
	Comment TEXT,
	ReviewDate DATE,
	FOREIGN KEY (BookID) REFERENCES Book (BookID),
	FOREIGN KEY (MemberID) REFERENCES Member (MemberID)
);

CREATE TABLE CATEGORY (
	CategoryKey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
	CategoryID INT NOT NULL,
	CategoryName VARCHAR(255) NOT NULL
);

CREATE TABLE MEMBER (
	Memberkey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
	MemberID INT NOT NULL,
	FirstName VARCHAR(255),
	LastName VARCHAR(255),
	Email VARCHAR(255) NOT NULL,
	PhoneNumber VARCHAR(50),
	MembershipDate DATE NOT NULL
);

CREATE TABLE RESERVATIONFACT (
	ReservationID INT PRIMARY KEY,
	BookKey INT,
	MemberKey INT,
	AuthorKey INT,
	CalendarKey INT,
	FOREIGN KEY (BookKey) REFERENCES BOOK(BookKey),
	FOREIGN KEY (MemberKey) REFERENCES MEMBER(MemberKey),
	FOREIGN KEY (AuthorKey) REFERENCES AUTHOR(AuthorKey),
	FOREIGN KEY (CalendarKey) REFERENCES CALENDAR(CalendarKey)
);

CREATE TABLE REVIEWFACT (
	ReviewID INT PRIMARY KEY,
	BookKey INT,
	MemberKey INT,
	AuthorKey INT,
	CalendarKey INT,
	Rating INT,
	Comment VARCHAR(MAX),
	FOREIGN KEY (BookKey) REFERENCES BOOK(BookKey),
	FOREIGN KEY (MemberKey) REFERENCES MEMBER(MemberKey),
	FOREIGN KEY (AuthorKey) REFERENCES AUTHOR(AuthorKey),
	FOREIGN KEY (CalendarKey) REFERENCES CALENDAR(CalendarKey)
);

CREATE TABLE AUTHOR (
	AuthorKey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
  	AuthorID INT NOT NULL,
	Name VARCHAR(255) NOT NULL,
	Nationality VARCHAR(100) NOT NULL,
	BirthYear INT
);

CREATE TABLE BOOK (
	BookKey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
	BookID INT NOT NULL,
	Title VARCHAR(255)NOT NULL,
	Genre VARCHAR(100) NOT NULL,
	PublishedYear INT NOT NULL,
	ISBN VARCHAR(50) NOT NULL
);

CREATE TABLE BORROWFACT (
	BorrowID INT PRIMARY KEY,
	BookKey INT,
	MemberKey INT,
	AuthorKey INT,
	CalendarKey INT,
	BorrowDuration INT,
	ReturnedLate BIT,
	FOREIGN KEY (BookKey) REFERENCES BOOK(BookKey),
	FOREIGN KEY (MemberKey) REFERENCES MEMBER(MemberKey),
	FOREIGN KEY (AuthorKey) REFERENCES AUTHOR(AuthorKey),
	FOREIGN KEY (CalendarKey) REFERENCES CALENDAR(CalendarKey)
);

CREATE TABLE BRIDGEBOOKCATEGORYFACT (
	BookKey INT,
	CategoryKey INT,
	FOREIGN KEY (CategoryKey) REFERENCES CATEGORY(CategoryKey),
	FOREIGN KEY (BookKey) REFERENCES BOOK(BookKey)
);

CREATE TABLE CALENDAR(
	CalendarKey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
	FullDate DATE NOT NULL,
	Year INT NOT NULL,
	Month INT NOT NULL,
	DayOfMonth INT NOT NULL,
	DayOfWeek VARCHAR(10) NOT NULL,
	Quarter INT NOT NULL,
	CHECK (Quarter > 0 AND Quarter <= 4),
	CHECK (Month > 0 AND Month <= 12),
	CHECK (DayOfMonth > 0 AND DayOfMonth <= 31)
);
