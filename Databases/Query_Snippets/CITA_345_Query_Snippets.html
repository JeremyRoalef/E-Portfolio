SQL Schema Setup



CREATE TABLE Author (
	AuthorID INT PRIMARY KEY, Name VARCHAR(255),
	Nationality VARCHAR(100),
	BirthYear INT
);

CREATE TABLE Book (
	BookID INT PRIMARY KEY, Title VARCHAR(255),
	AuthorID INT,
	Genre VARCHAR(100),
	PublishedYear INT,
	ISBN VARCHAR(50),
	FOREIGN KEY (AuthorID) REFERENCES Author (AuthorID)
);

CREATE TABLE BookCategory (
	BookID INT,
	CategoryID INT,
	PRIMARY KEY (BookID, CategoryID),
	FOREIGN KEY (BookID) REFERENCES Book (BookID),
	FOREIGN KEY (CategoryID) REFERENCES Category(CategoryID)
);

CREATE TABLE Borrowing (
	BorrowID INT PRIMARY KEY,
	BookID INT,
	MemberID INT,
	BorrowDate DATE,
	ReturnDate DATE,
	FOREIGN KEY (BookID) REFERENCES Book (BookID),
	FOREIGN KEY (MemberID) REFERENCES Member (MemberID)
);

CREATE TABLE Category (
	CategoryID INT PRIMARY KEY,
	CategoryName VARCHAR(100)
);

CREATE TABLE Member (
	MemberID INT PRIMARY KEY,
	FirstName VARCHAR(255),
	LastName VARCHAR(255),
	Email VARCHAR(255),
	PhoneNumber VARCHAR(50),
	MembershipDate DATE
);

CREATE TABLE Reservation (
	ReservationID INT PRIMARY KEY,
	BookID INT,
	MemberID INT,
	ReservationDate DATE,
	FOREIGN KEY (BookID) REFERENCES Book (BookID),
	FOREIGN KEY (MemberID) REFERENCES Member (MemberID)
);

CREATE TABLE Review (
	ReviewID INT PRIMARY KEY,
	BookID INT,
	MemberID INT,
	Rating INT,
	Comment TEXT,
	ReviewDate DATE,
	FOREIGN KEY (BookID) REFERENCES Book (BookID),
	FOREIGN KEY (MemberID) REFERENCES Member (MemberID)
);

SQL Schema Setup



CREATE TABLE CATEGORY (
	CategoryKey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
	CategoryID INT NOT NULL,
	CategoryName VARCHAR(255) NOT NULL
);

CREATE TABLE MEMBER (
	Memberkey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
	MemberID INT NOT NULL,
	FirstName VARCHAR(255),
	LastName VARCHAR(255),
	Email VARCHAR(255) NOT NULL,
	PhoneNumber VARCHAR(50),
	MembershipDate DATE NOT NULL
);

CREATE TABLE RESERVATIONFACT (
	ReservationID INT PRIMARY KEY,
	BookKey INT,
	MemberKey INT,
	AuthorKey INT,
	CalendarKey INT,
	FOREIGN KEY (BookKey) REFERENCES BOOK(BookKey),
	FOREIGN KEY (MemberKey) REFERENCES MEMBER(MemberKey),
	FOREIGN KEY (AuthorKey) REFERENCES AUTHOR(AuthorKey),
	FOREIGN KEY (CalendarKey) REFERENCES CALENDAR(CalendarKey)
);

CREATE TABLE REVIEWFACT (
	ReviewID INT PRIMARY KEY,
	BookKey INT,
	MemberKey INT,
	AuthorKey INT,
	CalendarKey INT,
	Rating INT,
	Comment VARCHAR(MAX),
	FOREIGN KEY (BookKey) REFERENCES BOOK(BookKey),
	FOREIGN KEY (MemberKey) REFERENCES MEMBER(MemberKey),
	FOREIGN KEY (AuthorKey) REFERENCES AUTHOR(AuthorKey),
	FOREIGN KEY (CalendarKey) REFERENCES CALENDAR(CalendarKey)
);

CREATE TABLE AUTHOR (
	AuthorKey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
  	AuthorID INT NOT NULL,
	Name VARCHAR(255) NOT NULL,
	Nationality VARCHAR(100) NOT NULL,
	BirthYear INT
);

CREATE TABLE BOOK (
	BookKey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
	BookID INT NOT NULL,
	Title VARCHAR(255)NOT NULL,
	Genre VARCHAR(100) NOT NULL,
	PublishedYear INT NOT NULL,
	ISBN VARCHAR(50) NOT NULL
);

CREATE TABLE BORROWFACT (
	BorrowID INT PRIMARY KEY,
	BookKey INT,
	MemberKey INT,
	AuthorKey INT,
	CalendarKey INT,
	BorrowDuration INT,
	ReturnedLate BIT,
	FOREIGN KEY (BookKey) REFERENCES BOOK(BookKey),
	FOREIGN KEY (MemberKey) REFERENCES MEMBER(MemberKey),
	FOREIGN KEY (AuthorKey) REFERENCES AUTHOR(AuthorKey),
	FOREIGN KEY (CalendarKey) REFERENCES CALENDAR(CalendarKey)
);

CREATE TABLE BRIDGEBOOKCATEGORYFACT (
	BookKey INT,
	CategoryKey INT,
	FOREIGN KEY (CategoryKey) REFERENCES CATEGORY(CategoryKey),
	FOREIGN KEY (BookKey) REFERENCES BOOK(BookKey)
);

CREATE TABLE CALENDAR(
	CalendarKey INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
	FullDate DATE NOT NULL,
	Year INT NOT NULL,
	Month INT NOT NULL,
	DayOfMonth INT NOT NULL,
	DayOfWeek VARCHAR(10) NOT NULL,
	Quarter INT NOT NULL,
	CHECK (Quarter > 0 AND Quarter <= 4),
	CHECK (Month > 0 AND Month <= 12),
	CHECK (DayOfMonth > 0 AND DayOfMonth <= 31)
);


/*
ETL Infastructure for inserting into Review Fact Table:
1) Retrieve data from the operational database in the given order:
	a. ReviewID
	b. BookID
	c. MemberID
	d. AuthorID
	e. Rating
	f. Comment
2) Retrieve the key values from the analytical database and join them with the corresponding IDs.
3) Insert the values with the keys in the given order:
	a. ReviewID
	b. BookKey
	c. MemberKey
	d. AuthorKey
	e. CalendarKey (linked to BorrowDate)
	d. Rating
	e. Comment
4) Prevent existing data from repopulating. Use EXCEPT to remove existing data.
*/

INSERT INTO [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.REVIEWFACT
	--SELECTION
	SELECT 
		opREV.ReviewID,
		anaB.BookKey, 
		anaM.Memberkey, 
		anaAU.AuthorKey, 
		anaCAL.CalendarKey, 
		opREV.Rating,
		opREV.Comment
	FROM [CITA 345 - Project 2 - Operational DB LIbrary Management System].dbo.Review opREV
	--Grab reference to author
	JOIN [CITA 345 - Project 2 - Operational DB LIbrary Management System].dbo.Book opB
		ON opB.BookID = opREV.BookID
	JOIN [CITA 345 - Project 2 - Operational DB LIbrary Management System].dbo.Author opAU
		ON opAU.AuthorID = opB.AuthorID
	--Retrieve key values from warehouse
	JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.BOOK anaB
		ON anaB.BookID = opB.BookID
	JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.MEMBER anaM
		ON anaM.MemberID = opREV.MemberID
	JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.AUTHOR anaAU
		ON anaAU.AuthorID = opAU.AuthorID
	JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.CALENDAR anaCAL
		ON anaCAL.FullDate = opREV.ReviewDate
	--Exclusion of existing data
	EXCEPT
	SELECT 
		anaREV.ReviewID, 
		anaREV.BookKey, 
		anaREV.MemberKey, 
		anaREV.AuthorKey, 
		anaREV.CalendarKey, 
		anaREV.Rating, 
		anaREV.Comment
	FROM [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.REVIEWFACT anaREV

/*
ETL Process for Author Dimension:

1) Get the required information from the operational database in the given order:
	a. author ID
	b. Name
	c. Nationality
	d. Birth Year

2) After retrieving the information, you need to insert it into the author dimension of the analytical
   database. Also, you need to FILTER out rows that already exist in the analytical database.
   Since the data in the analytical database are technically a subset of the data in the operational
   database, you can use EXCEPT to exclude certain rows from being inserted (similar to NOT IN operator).
   This should generally work for the purposes of this data warehouse, but is not the end all be all solution.
*/

--opA stands for operational DB author table
--anaA stands for analytical DB author dimension
INSERT INTO [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.AUTHOR (AuthorID, Name, Nationality, BirthYear)
	SELECT opA.AuthorID, opA.name, opA.Nationality, opA.BirthYear
		FROM [CITA 345 - Project 2 - Operational DB LIbrary Management System].dbo.Author opA
		EXCEPT
		SELECT anaA.authorID, anaA.name, anaA.nationality, anaA.BirthYear
		FROM [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.AUTHOR anaA

/*
ETL Process for inserting new books into the warehouse:

1) Select columns in the following order:
	a. BookID
	b. Title
	c. Genre
	d. PublishedYear
	e. ISBN
2) Use an insert statement to insert into the analytical DB book dimension, and use EXCEPT to exclude entries that
   already exist in the dimension
*/

INSERT INTO [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.BOOK
	SELECT opB.BookID, opb.Title, opb.Genre, opb.PublishedYear, opb.ISBN
		FROM [CITA 345 - Project 2 - Operational DB LIbrary Management System].dbo.Book opB
		EXCEPT
		SELECT anaB.BookID, anaB.Title, anaB.Genre, anaB.PublishedYear, anaB.ISBN
		FROM [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.BOOK anaB

/*
ETL Infastructure for inserting into Borrow Fact Table:
1) Retrieve data from the operational database in the given order:
	a. BorrowID
	b. BookID
	c. MemberID
	d. AuthorID
	e. BorrowDate
2) Retrieve the key values from the analytical database and join them with the corresponding IDs.
3) Insert the values with the keys in the given order:
	a. BorrowID
	b. BookKey
	c. MemberKey
	d. AuthorKey
	e. CalendarKey (linked to BorrowDate)
	d. BorrowDuration (calculated by ReturnDate-BorrowDate)
	e. ReturnedLate (if ReturnDate-BorrowDate > 14, then value = 1. else, value = 0)
4) Prevent existing data from repopulating. Use EXCEPT to remove existing data.
*/

INSERT INTO [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.BORROWFACT
	--SELECTION
	SELECT 
		opBOR.BorrowID, 
		anaB.BookKey, 
		anaM.Memberkey, 
		anaAU.AuthorKey, 
		anaCAL.CalendarKey, 
		DATEDIFF(DAY, opBOR.BorrowDate, opBOR.ReturnDate) AS BorrowDuration, 
		CAST(CASE WHEN DATEDIFF(DAY,opBor.BorrowDate, opBOR.ReturnDate) > 14 THEN 1 ELSE 0 END AS bit) AS ReturnedLate
	FROM [CITA 345 - Project 2 - Operational DB LIbrary Management System].dbo.Borrowing opBOR
	--Grab reference to author
	JOIN [CITA 345 - Project 2 - Operational DB LIbrary Management System].dbo.Book opB
		ON opB.BookID = opBOR.BookID
	JOIN [CITA 345 - Project 2 - Operational DB LIbrary Management System].dbo.Author opAU
		ON opAU.AuthorID = opB.AuthorID
	--Retrieve key values from warehouse
	JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.BOOK anaB
		ON anaB.BookID = opB.BookID
	JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.MEMBER anaM
		ON anaM.MemberID = opBOR.MemberID
	JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.AUTHOR anaAU
		ON anaAU.AuthorID = opAU.AuthorID
	JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.CALENDAR anaCAL
		ON anaCAL.FullDate = opBOR.BorrowDate
	--Exclusion of existing data
	EXCEPT
	SELECT 
		anaBOR.BorrowID, 
		anaBOR.BookKey, 
		anaBOR.MemberKey, 
		anaBOR.AuthorKey, 
		anaBOR.CalendarKey, 
		anaBOR.BorrowDuration, 
		anaBOR.ReturnedLate
	FROM [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.BORROWFACT anaBOR


/*
ETL Process for inserting into BridgeBookCategory Fact Table

1) Retrieve the bookID and CategoryID from each row in given order:
	a. BookID
	b. CategoryID
2) Join the BookID onto the book dimension where op.BookID = analytical.BookID, and get THAT BookKey.
3) Join the CategoryID onto the Category Dimension where op.CategoryID = analytical.CategoryID, and get THAT CategoryKey
4) Insert the BookID-CategoryID key values as a new entry into the BridgeBookCategory Fact Table in the given order:
	a. BookKey
	b. CategoryKey
5) Exclude exising rows from the insertion process. Use EXCEPT
*/

--Insertion into BRIDGEBOOKCATEGORYFACT
INSERT INTO [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.BRIDGEBOOKCATEGORYFACT
	--Row Selection
	SELECT anaBook.BookKey, anaCategory.CategoryKey
		FROM [CITA 345 - Project 2 - Operational DB LIbrary Management System].dbo.BookCategory opBookCategory
		JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.BOOK anaBook
		ON anaBook.BookID = opBookCategory.BookID
		JOIN [CITA 345 - Project 2 - Analytical DB Library Management System].dbo.CATEGORY anaCategory
		ON anaCategory.CategoryID = opBookCategory.CategoryID

		EXCEPT
		SELECT anaBBCF.BookKey, anaBBCF.CategoryKey
