University Course Management Sysetm Database



CREATE TABLE building(	
	bid INT NOT NULL,  
	bname VARCHAR(24) NOT NULL,   
	PRIMARY KEY (bid)
	);    

CREATE TABLE classroom(
	roomno INT NOT NULL,  
	bid INT NOT NULL,  
	FOREIGN KEY (bid) REFERENCES building(bid),
	PRIMARY KEY (roomno, bid) 
);  
 
CREATE TABLE department(	
	did INT NOT NULL,  
	dname VARCHAR(24) NOT NULL,  
	PRIMARY KEY (did) 
);  
  

CREATE TABLE major(
	mid INT NOT NULL,  
	did INT NOT NULL,  
	mname VARCHAR(24) NOT NULL,  
	FOREIGN KEY (did) REFERENCES department(did), 
	PRIMARY KEY (mid) 
);  

CREATE TABLE course(
	ccode INT NOT NULL,  
	ctitle VARCHAR(24) NOT NULL,  
	cdesc VARCHAR(24) NOT NULL,  
	ccredits NUMERIC(1,0) NOT NULL,  
	did INT NOT NULL,	  
	FOREIGN KEY (did) REFERENCES department(did), 
	PRIMARY KEY (ccode)
);   

CREATE TABLE requirement(	
	ccode INT NOT NULL,  
	mid INT NOT NULL,	  
	FOREIGN KEY (ccode) REFERENCES course(ccode), 
	FOREIGN KEY (mid) REFERENCES major(mid), 
	PRIMARY KEY (ccode, mid) 
);   

CREATE TABLE student(
	stid INT NOT NULL,  
	stfirstname VARCHAR(24) NOT NULL, 
	stlastname VARCHAR(24) NOT NULL,  
	stemail VARCHAR(24) NOT NULL,  
	stdob DATE NOT NULL,  
	mid INT NOT NULL,  
	FOREIGN KEY (mid) REFERENCES major(mid),	  
	PRIMARY KEY (stid) 
);   

CREATE TABLE falculty(	
	fid INT NOT NULL,  
	ffirstname VARCHAR(24) NOT NULL,  
	flastname VARCHAR(24) NOT NULL,  
	frank VARCHAR(24) NOT NULL,  
	did INT NOT NULL,  
	FOREIGN KEY (did) REFERENCES department(did),  
	PRIMARY KEY (fid)
);    

CREATE TABLE section(
	csid INT NOT NULL,  
	cssemester VARCHAR(24) NOT NULL,  
	csyear INT NOT NULL,  
	roomno INT NOT NULL,  
	bid INT NOT NULL,  
	ccode INT NOT NULL,  
	fid INT NOT NULL,  
	FOREIGN KEY (roomno, bid) REFERENCES classroom(roomno, bid),  
	FOREIGN KEY (ccode) REFERENCES course(ccode),  
	FOREIGN KEY (fid) REFERENCES falculty(fid),  
	PRIMARY KEY (csid, cssemester, csyear, ccode) 
);  

CREATE TABLE enrollment(
	edate DATE NOT NULL,   
	egrade INT NOT NULL,   
	stid INT NOT NULL,   
	csid INT NOT NULL,  
	ccode INT NOT NULL,   
	cssemester VARCHAR(24) NOT NULL,   
	csyear INT NOT NULL,   
	FOREIGN KEY (stid) REFERENCES student(stid), 
	FOREIGN KEY (csid, cssemester, csyear, ccode) REFERENCES section(csid, cssemester, csyear, ccode),   
	PRIMARY KEY (stid, csid, cssemester, csyear, ccode) 
);

/*
Display the average grade for a specific class section.

This query will display the average grade in the class PSYCH 101 Section 1 in Fall 2024
*/

SELECT c.ctitle, AVG(e.egrade) AS ClassAverageGrade
	FROM enrollment e, course c
	WHERE e.ccode = 2 AND e.csid = 1 AND e.cssemester = 'FALL' AND e.csyear = 2024 AND c.ccode = e.ccode
	GROUP BY e.ccode, e.csid, e.cssemester, e.csyear, c.ctitle



/*
Identify students who are enrolled in more than three courses

(No student enrolled in more than 3 courses. Result will be nothing)
*/
SELECT st.stfirstname AS FirstName, st.stlastname AS LastName
	FROM student st
	JOIN enrollment e
	ON st.stid = e.stid
	WHERE (
		SELECT COUNT(stid) 
		FROM enrollment 
		GROUP BY stid 
		HAVING st.stid = stid
		) > 3



/*
Generate a report of all classrooms being used in a specific semester

This query shows all classrooms being used in spring 2026 semester
*/
SELECT cs.cssemester AS Semester, cs.csyear AS Year, b.bname AS Building, cr.roomno AS RoomNum
	FROM section cs
	JOIN classroom cr
	ON cs.roomno = cr.roomno AND cs.bid = cr.bid
	JOIN building b
	ON cr.bid = b.bid
	WHERE cs.cssemester = 'SPRING' AND cs.csyear = 2026



/*
List all faculty members teaching in a specific semester

This query will list all falculty members teaching in the spring 2026 semester
*/
SELECT cs.cssemester AS Semester, cs.csyear AS Year, f.ffirstname AS FirstName, f.flastname AS LastName
	FROM section cs
	JOIN falculty f
	ON cs.fid = f.fid
	WHERE cs.cssemester = 'SPRING' AND cs.csyear = 2026
