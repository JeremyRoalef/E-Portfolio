PlayerController.cs

Handles player movement input and applies velocity to the character’s Rigidbody2D. This script lets the player move left and right using the arrow keys or A/D keys. 


using UnityEngine;

public class PlayerController : MonoBehaviour
{
    public float speed = 5f;
    private Rigidbody2D rb;

    void Start()
    {
        rb = GetComponent();
    }

    void Update()
    {
        float moveX = Input.GetAxis("Horizontal");
        rb.velocity = new Vector2(moveX * speed, rb.velocity.y);
    }
}

EnemyAI.cs

Makes an enemy follow the player using simple vector math. The enemy constantly moves toward the player’s position at a fixed speed. 


using UnityEngine;

public class EnemyAI : MonoBehaviour
{
    public Transform player;
    public float chaseSpeed = 2f;

    void Update()
    {
        if (player != null)
        {
            transform.position = Vector2.MoveTowards(
                transform.position,
                player.position,
                chaseSpeed * Time.deltaTime
            );
        }
    }
}
